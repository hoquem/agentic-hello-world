def run(self):
    # 1) Build the request from the conversation so far,
    #    and show the UI exactly what we send.
    request = build_request(self.messages, self.model)
    yield {"type": "sent", "request": request}

    # 2) Stream the reply back, piece by piece.
    #    self._chat_fn is what talks to Ollama.
    assembled = ""
    for chunk in self._chat_fn(request):
        content = chunk["message"]["content"]
        # some chunks are "thinking" only — no text
        if content:
            assembled += content
        # but always show the raw chunk — nothing hidden
        yield {"type": "received_chunk", "raw": chunk}

    # 3) Save the reply, then tell the UI to show it.
    self.messages.append(
        {"role": "assistant", "content": assembled}
    )
    yield {"type": "display", "content": assembled}
The model is stateless. Everything an "agent" does lives in code this simple — the harness, not the model.
